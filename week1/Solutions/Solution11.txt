Problem11: Binary Search

input: Sorted array A of size N, target x

output: Index of x or -1

constraints: 1 ≤ N ≤ 2*10^5, O(log N)


class Solution {
    public int search(int[] nums, int target) {
        int i=0,j=nums.length-1;
        while(i<=j){
            int mid = (i +j)/2;
            if(nums[mid]==target){
                return mid;
            }
            else if(nums[mid]>target){
                j=mid-1;
            }
            else{
               i=mid+1;
            }
        }
        return -1;
    }
}
