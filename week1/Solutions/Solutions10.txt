Problem10: Maximum Subarray Sum

input: Array A of size N

output: Maximum subarray sum

constraints: 1 ≤ N ≤ 2*10^5, -10^9 ≤ A[i] ≤ 10^9

//Algorithm
class Solution {
    public int maxSubArray(int[] nums) {
        int sum = nums[0];
        int currSum = nums[0];
        for(int i=1;i<nums.length;i++){
            if(currSum <0){
                currSum=0;
            }
            currSum=nums[i]+currSum;
            
            if(currSum>sum){
                sum = currSum;
            }
        }
        return sum;
    }
}
