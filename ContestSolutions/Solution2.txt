// Basic Approach 
class Solution {
    static boolean isPrime(int n) {
        if(n==2 || n==3 ){
            return true;
        }
        int c=0;
        for(int i=2;i<=n/2;i++){
            if(n%i==0){
                c++;
            }
        }
        return c==2;
    }
}

// Optimized
import java.util.*;
class Main{
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if(isPrime(n)){
            System.out.print("Yes");
        }
        else{
            System.out.print("No");
        }
    }
    static boolean isPrime(int n) {
        if(n==1){
           return false;
       }
       int c=0;
       for(int i=2;i*i<=n;i++){
           if(n%i==0){
            c++;
           }
       }
       return c==1;
    }

}
