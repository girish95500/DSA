// Basic Approach 
class Solution {
    static boolean isPrime(int n) {
        if(n==2 || n==3 ){
            return true;
        }
        int c=0;
        for(int i=2;i<=n/2;i++){
            if(n%i==0){
                c++;
            }
        }
        return c==2;
    }
}
