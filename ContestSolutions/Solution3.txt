Q3. Prime Sum Pair

Problem
Given an integer n, determine whether it can be expressed as the sum of two prime numbers.

Input
A single integer n

Output
Print "YES" if n can be written as p1 + p2 where both p1 and p2 are prime numbers.
Otherwise, print "NO".

Constraints
2 ≤ n ≤ 10^7

Example
Input: 34
Output: YES

Input: 11
Output: NO


// Basic Approach
public class Main {
    public static void main(String[] args) {
        
        int n = 11;

        int[] arr = new int[n + 1]; 

        for (int i = 2; i * i <= n; i++) {
            if (arr[i] == 0) {
                for (int j = i * i; j <= n; j += i) {
                    arr[j] = 1; 
                }
            }
        }
          boolean flag = false;
        for (int i = 2; i < n; i++) {
            if (arr[i] == 0 && arr[n-i]==0 ) {
                System.out.print("n",arr[n]);
                System.out.print("i",arr[i]);
                 flag = true;
                 break;
            }
        }
        if(flag){
            System.out.print("Yes");
        }
        else{
            System.out.print("No");
        }
        
    }
}
